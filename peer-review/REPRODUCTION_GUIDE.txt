git clone https://github.com/MettaMazza/ernos-labs-sft-platform.git sft-v3-review
git -C sft-v3-review checkout --detach 76f1b34e9e9682989adcbb088797b4ae4ce4fcf1
git clone https://github.com/MettaMazza/SFT-v4-LEAN-VERIFIED.git sft-v4-review
git -C sft-v4-review checkout --detach 43a7295f5d39a1df4ca25b2c1e9b8a21164d55ee
