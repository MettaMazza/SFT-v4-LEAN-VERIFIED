import SFT.Physics.FreePhase
namespace SFT.Physics.FreePhaseWitnesses
open FreePhase
structure Witness where
 name : String
 statement : String
 passed : Bool
def original := freeParticlePhase (.fraction ⟨1,3,by decide,by decide⟩) (.fraction ⟨1,4,by decide,by decide⟩)
def dispersionCheck : Bool := match original with
 | .error _ => false
 | .ok t => !t.dispersion.blt (⟨1,2,by decide,by decide⟩ : Rat) && !(⟨1,2,by decide,by decide⟩ : Rat).blt t.dispersion
def routeCheck : Bool := match original with
 | .error _ => false
 | .ok t => !t.phaseAfter.blt (⟨5,6,by decide,by decide⟩ : Rat) && !(⟨5,6,by decide,by decide⟩ : Rat).blt t.phaseAfter
def witnesses : List Witness := [WITNESS_ROWS]
theorem dispersionCheckPassed : dispersionCheck=true := by unfold dispersionCheck original;rw [originalAssembly];rfl
theorem routeCheckPassed : routeCheck=true := by unfold routeCheck original;rw [originalAssembly];rfl
theorem allWitnessesPass : witnesses.all Witness.passed=true := by
 change (dispersionCheck && routeCheck && true)=true
 rw [dispersionCheckPassed,routeCheckPassed]
 rfl
end SFT.Physics.FreePhaseWitnesses
